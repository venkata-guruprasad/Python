# Check the number is armstrong number or not
n = 371
temp = n
sum = 0
ord = len(str(n))
for i in range(ord):
    d = n%10
    sum = sum+d**ord
    n//=10
print(sum)

if sum == temp:
    print("Armstrong Number")
else:
    print("Not an Armstrong Number")

output:
371
Armstrong Number

# separate even and odd elements from given list
l = [1,2,3,4,5,6,7,8,9]
even =[]
odd = []
for i in l:
    if i%2 == 0:
        even.append(i)
    else:
        odd.append(i)
print("Even :",even)
print("Odd :",odd)

output:

Even : [2, 4, 6, 8]
Odd : [1, 3, 5, 7, 9]
