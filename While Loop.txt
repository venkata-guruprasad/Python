# Write a Python program to print each digit of a number separately.

n = int(input("Enter a number: "))

while n > 0:
    d = n % 10
    print(d,end=' ')
    n //= 10

output:  

Enter a number: 1234
4 3 2 1

# Write a Python program to find the sum of digits of a given number.
n = int(input("Enter a number: "))
sum = 0
while n > 0:
    d = n % 10
    sum += d
    n //= 10
print(sum)

output:

Enter a number: 1234
10

# print Smallest digit from the given number
n = int(input('Enter a number : '))
smallest = 9
while n>0:
    d = n%10
    if d<smallest:
        smallest = d
    n//=10
print(smallest)

output:

Enter a number : 4561
1


# armstrong number 
n = 371
temp = n
sum = 0
ord = len(str(n))
while n>0:
    d = n%10
    sum= sum+d**ord
    n//=10
print(sum)
if temp == sum:
    print('Armstrong Number',temp,sum)
else:
    print('Not an Armstrong number')

output:

371
Armstrong Number 371 371


# Write a Python program to reverse a given number.
n = int(input('Enter a Number : '))
rev = 0
while n>0:
    d = n%10
    rev = rev*10 + d
    n//=10
print(rev)

output : 
Enter a Number : 2131
1312
