# Find the average of numbers from 1 to N.
n = int(input('Enter a Number : '))
sum = 0
i = 1
while i<=n:
    sum+=i
    i+=1
avg = sum/n
print('Average :',avg)

output:

Enter a Number : 5
Average : 3.0

# Find the sum of numbers divisible by 3.
n = int(input('Enter a Number : '))
sum = 0
while n>0:
    d = n%10
    sum = sum+d
    n//=10
if sum % 3 ==0:
    print('Sum of numbers divisible by 3',sum)
else:
    print('Sum of numbers not divisible by 3')

output:

Enter a Number : 123
Sum of numbers divisible by 3 : 6
